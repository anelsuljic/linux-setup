# Installation and removal of packages

INSTALL_LIST="$HOME/<linux-setup-dir>/setups/setup-apps/install-list.txt"
INSTALL_LIST_TMP="$HOME/<linux-setup-dir>/setups/setup-apps/install-list.tmp"

install_pkg() {
    for pkg in "$@"; do                 
        if pacman -Si "$pkg" > /dev/null 2>&1 || \
           yay -Si "$pkg" > /dev/null 2>&1; then

            if pacman -Si "$pkg" > /dev/null 2>&1; then
                sudo pacman -S --needed --noconfirm "$pkg"
            else
                yay -S --needed --noconfirm "$pkg"
            fi
            
            if [ $? -eq 0 ]; then
                echo "Installation of $pkg succeeded."
                echo "$pkg" >> "$INSTALL_LIST"
            else
                echo "Error: installation of $pkg failed."
            fi
        else
            echo "Error: $pkg doesn't exist in pacman and yay."
        fi
    done
    sort -u -o "$INSTALL_LIST" "$INSTALL_LIST"
}

# ---- NVIDIA dGPU power (ASUS G513 hybrid) ----------------------------------
# "Off" = PCI D3cold (~0W) via runtime PM. A firmware hard-disable can't run
# while Hyprland holds the dGPU for HDMI, so D3cold is the real off.
dgpu() {                # status
    local d=/sys/bus/pci/devices/0000:01:00.0
    printf 'dGPU: power=%s runtime=%s control=%s\n' \
        "$(<"$d/power_state")" "$(<"$d/power/runtime_status")" "$(<"$d/power/control")"
}
dgpu-off() {            # run when you don't need the dGPU (no monitor, no GPU job)
    local d=/sys/bus/pci/devices/0000:01:00.0
    echo auto | sudo tee "$d/power/control" >/dev/null
    sleep 4
    if [ "$(<"$d/power_state")" = D3cold ]; then
        echo "dGPU powered down (D3cold)."
    else
        echo "dGPU still on — an external display or a GPU job is using it (or give it a few seconds; recheck with 'dgpu')."
    fi
}
